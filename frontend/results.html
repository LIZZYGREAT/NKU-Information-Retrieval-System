document.addEventListener('DOMContentLoaded', () => { const urlParams = new URLSearchParams(window.location.search); const currentQuery = urlParams.get('q'); const currentType = urlParams.get('type') || 'site'; let currentPage = parseInt(urlParams.get('page')) || 1; const topInput = document.getElementById('top-search-input'); const searchBtn = document.getElementById('top-search-btn'); const container = document.getElementById('results-container'); const stats = document.getElementById('result-stats'); const pagination = document.getElementById('pagination-container'); // 初始化搜索框状态 if (currentQuery) { topInput.value = currentQuery; fetchResults(); } // 重新检索绑定 searchBtn.addEventListener('click', () => { const newQuery = topInput.value.trim(); if (newQuery) { window.location.href = `results.html?q=${encodeURIComponent(newQuery)}&type=${currentType}&page=1`; } }); async function fetchResults() { container.innerHTML = '

检索中...

'; const user = window.auth.getUser(); try { // 调用后端的 POST /api/search 接口 const response = await window.apiClient.post('/search', { query_text: currentQuery, search_type: currentType, user_id: user ? user.user_id : null, page: currentPage }); const data = response.data; renderResults(data.results, data.total_hits); renderPagination(data.total_hits); } catch (error) { container.innerHTML = `

请求失败: ${error.message}

`; } } function renderResults(results, totalHits) { stats.textContent = `共找到 ${totalHits} 条结果`; container.innerHTML = ''; if (!results || results.length === 0) { container.innerHTML = '

未找到匹配内容。

'; return; } results.forEach(item => { const div = document.createElement('div'); div.className = 'result-item'; // 构建网页快照的 URL const snapshotUrl = `http://localhost:8000/api/snapshot?url=${encodeURIComponent(item.url)}`; // innerHTML 注入 Elasticsearch 的高亮 标签 div.innerHTML = ` ${item.title || '无标题'}
URL: ${item.url} [查看网页快照]
${item.highlight}
`; container.appendChild(div); }); } function renderPagination(totalHits) { pagination.innerHTML = ''; const pageSize = 10; const totalPages = Math.ceil(totalHits / pageSize); if (totalPages <= 1) return; if (currentPage > 1) { const prev = document.createElement('button'); prev.textContent = '上一页'; prev.className = 'btn'; prev.onclick = () => changePage(currentPage - 1); pagination.appendChild(prev); } if (currentPage < totalPages) { const next = document.createElement('button'); next.textContent = '下一页'; next.className = 'btn'; next.onclick = () => changePage(currentPage + 1); pagination.appendChild(next); } } function changePage(page) { window.location.href = `results.html?q=${encodeURIComponent(currentQuery)}&type=${currentType}&page=${page}`; } });
